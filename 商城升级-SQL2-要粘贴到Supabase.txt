商城升级 SQL · 第 2 批（柚子专属 / 多包裹 / 收藏）
· 一次性粘贴到 Supabase SQL Editor 运行即可
· 全部为「新增」表/函数/策略：不改、不删任何现有表和数据，可重复运行
· 与第 1 批互不依赖，先后皆可
· 注意：测试站与正式站共用同一个 Supabase 项目，本批因为只做新增，
  对正式站完全无影响；「删除 site_settings 旧键」的收尾一步在正式版
  发布之后才做，本批不包含。

-- ═══════════ 第 A 步：柚子专属（数据层）═══════════
-- 专属内容数据表：kind = bjds / cards / works
create table if not exists public.yuki_exclusive (
  kind text primary key,
  payload jsonb not null default '[]'::jsonb,
  updated_at timestamptz not null default now()
);

-- 白名单表：不出任何 API 策略（REST 直读/直写一律拒绝），只在 SQL 编辑器维护
create table if not exists public.yuki_exclusive_emails (
  email text primary key
);
insert into public.yuki_exclusive_emails (email)
values ('chengyuki119@gmail.com')
on conflict (email) do nothing;

alter table public.yuki_exclusive        enable row level security;
alter table public.yuki_exclusive_emails enable row level security;

-- 专属数据：仅白名单邮箱可读，管理员（shop_admins）可写（后台双写用）
drop policy if exists yx_read on public.yuki_exclusive;
create policy yx_read on public.yuki_exclusive for select
  using (public.yuki_is_exclusive());
drop policy if exists yx_write on public.yuki_exclusive;
create policy yx_write on public.yuki_exclusive for all
  using (public.shop_is_admin()) with check (public.shop_is_admin());

-- 白名单判断函数（security definer 才能读白名单表）
create or replace function public.yuki_is_exclusive() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.yuki_exclusive_emails e
    where lower(e.email) = lower(coalesce(current_setting('request.jwt.claim.email', true), ''))
  );
$$;

-- 读取 RPC：非白名单一律拒绝；ping 用于前端探测登录身份
create or replace function public.yuki_exclusive_get(p_kind text) returns jsonb
language plpgsql stable security definer set search_path = public as $$
begin
  if not public.yuki_is_exclusive() then
    raise exception '此内容仅限柚子专属账号访问';
  end if;
  if p_kind = 'ping' then
    return jsonb_build_object('ok', true, 'email',
      coalesce(current_setting('request.jwt.claim.email', true), ''));
  end if;
  if p_kind not in ('bjds','cards','works') then
    raise exception '未知的专属内容类型';
  end if;
  return coalesce((select payload from public.yuki_exclusive where kind = p_kind), '[]'::jsonb);
end $$;
grant execute on function public.yuki_exclusive_get(text) to authenticated;

-- 把 site_settings 里现有的三份数据复制过来（只复制一次，之后以后台保存为准）
insert into public.yuki_exclusive (kind, payload)
select 'bjds',  value from public.site_settings where key = 'bjds'
on conflict (kind) do nothing;
insert into public.yuki_exclusive (kind, payload)
select 'cards', value from public.site_settings where key = 'cards'
on conflict (kind) do nothing;
insert into public.yuki_exclusive (kind, payload)
select 'works', value from public.site_settings where key = 'works'
on conflict (kind) do nothing;

-- ═══════════ 第 B 步：订单多包裹 ════════════
create table if not exists public.shop_packages (
  id uuid primary key default gen_random_uuid(),
  order_id uuid not null references public.shop_orders(id) on delete cascade,
  carrier text not null default '',        -- UPS / USPS / FedEx / Other
  tracking text not null default '',
  track_url text not null default '',
  weight_oz numeric(10,1),
  shipped_at timestamptz,
  note text not null default '',
  created_at timestamptz not null default now()
);
alter table public.shop_packages enable row level security;

-- 买家只看自己订单的包裹；管理员全权
drop policy if exists spkg_select on public.shop_packages;
create policy spkg_select on public.shop_packages for select
  using (public.shop_is_admin() or exists (
    select 1 from public.shop_orders o where o.id = order_id and o.user_id = auth.uid()));
drop policy if exists spkg_admin on public.shop_packages;
create policy spkg_admin on public.shop_packages for all
  using (public.shop_is_admin()) with check (public.shop_is_admin());

-- ═══════════ 第 C 步：商品收藏 ════════════
create table if not exists public.shop_favorites (
  user_id uuid not null references auth.users(id) on delete cascade,
  product_id uuid not null references public.products(id) on delete cascade,
  created_at timestamptz not null default now(),
  primary key (user_id, product_id)
);
alter table public.shop_favorites enable row level security;
drop policy if exists sfav_own on public.shop_favorites;
create policy sfav_own on public.shop_favorites for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);
