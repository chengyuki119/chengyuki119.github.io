-- ═══════════ 第 0 步：管理员白名单（★改成你的后台登录邮箱★）═══════════
create table if not exists public.shop_admins (
  user_id uuid primary key references auth.users(id) on delete cascade,
  created_at timestamptz not null default now()
);
insert into public.shop_admins (user_id)
select id from auth.users where email = '你的后台邮箱@example.com'
on conflict do nothing;

-- 管理员判断函数（所有策略共用）
create or replace function public.shop_is_admin() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.shop_admins a where a.user_id = auth.uid());
$$;

-- ═══════════ 第 3 步：订单 ═══════════
create table if not exists public.shop_orders (
  id uuid primary key default gen_random_uuid(),
  order_no text not null unique,
  user_id uuid not null references auth.users(id) on delete cascade,
  email text not null default '',
  items jsonb not null default '[]',          -- 快照 [{id,name,image,sku,variant,price,qty,weight_oz}]
  subtotal numeric(10,2) not null default 0,
  discount numeric(10,2) not null default 0,
  coupon_code text not null default '',
  shipping_est numeric(10,2) not null default 0,   -- 下单时运费估算（快照）
  shipping_actual numeric(10,2),                   -- 后台录入的 Pirate Ship 实际运费
  service_fee numeric(10,2) not null default 2,    -- 服务费（后台可在 site_settings.shop_ship 改默认值）
  total numeric(10,2) not null default 0,          -- 应付 = subtotal - discount + (实际运费或估算) + service_fee
  status text not null default 'pending',
  -- pending 待付款截图 / reviewing 待审核付款 / rejected 凭证有误 / paid 已付款(待发货)
  -- shipped 已发货 / completed 已完成 / cancelled 已取消 / refunding 退款处理中 / refunded 已退款
  pay_url text not null default '',           -- 付款截图（私有桶 pay）
  pay_amount numeric(10,2),                   -- 买家填的付款金额
  pay_channel text not null default '',       -- 付款渠道
  pay_time text not null default '',          -- 付款时间（买家填）
  pay_note text not null default '',          -- 买家付款备注
  reject_reason text not null default '',     -- 凭证拒绝原因
  address jsonb not null default '{}'::jsonb, -- 下单地址快照
  note text not null default '',              -- 订单备注
  group_id uuid,                              -- 拼团订单关联
  client_token text unique,                   -- 防重复下单
  carrier text not null default '',           -- UPS / USPS / FedEx / Other
  tracking text not null default '',
  track_url text not null default '',
  shipped_at timestamptz,
  ship_note text not null default '',
  admin_note text not null default '',        -- 仅后台可见内部备注
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- ═══════════ 第 8 步：安全触发器（买家改不了金额/状态机/库存自动回补）═══════════
-- 买家更新订单：只允许 改付款凭证字段 + 状态机内合法流转；金额/物流等字段动不了
create or replace function public.shop_guard_order_update() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  if public.shop_is_admin() then
    NEW.updated_at = now();
    if OLD.status is distinct from NEW.status then
      insert into public.shop_order_logs (order_id, actor, action, detail)
      values (NEW.id, 'admin', 'status', OLD.status || ' → ' || NEW.status);
    end if;
    return NEW;
  end if;
  -- 买家：锁死金额与物流
  NEW.subtotal = OLD.subtotal; NEW.discount = OLD.discount; NEW.total = OLD.total;
  NEW.shipping_actual = OLD.shipping_actual; NEW.service_fee = OLD.service_fee;
  NEW.shipping_est = OLD.shipping_est; NEW.admin_note = OLD.admin_note;
  NEW.carrier = OLD.carrier; NEW.tracking = OLD.tracking; NEW.track_url = OLD.track_url;
  NEW.shipped_at = OLD.shipped_at; NEW.ship_note = OLD.ship_note;
  NEW.coupon_code = OLD.coupon_code; NEW.items = OLD.items; NEW.address = OLD.address;
  NEW.group_id = OLD.group_id; NEW.order_no = OLD.order_no;
  -- 状态机：买家只能 pending/reviewing/rejected → cancelled；rejected → reviewing（重传凭证）；shipped → completed
  if NEW.status is distinct from OLD.status then
    if not (
      (OLD.status in ('pending','reviewing','rejected') and NEW.status = 'cancelled') or
      (OLD.status = 'rejected' and NEW.status = 'reviewing') or
      (OLD.status = 'shipped' and NEW.status = 'completed')
    ) then
      raise exception '不允许的订单状态变更';
    end if;
    insert into public.shop_order_logs (order_id, actor, action, detail)
    values (NEW.id, 'buyer', 'status', OLD.status || ' → ' || NEW.status);
  end if;
  NEW.updated_at = now();
  return NEW;
end $$;
drop trigger if exists trg_shop_guard_order on public.shop_orders;
create trigger trg_shop_guard_order before update on public.shop_orders
  for each row execute function public.shop_guard_order_update();

-- 取消/退款时自动回补库存（只回补一次）
create or replace function public.shop_restore_stock() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  it jsonb; v_id uuid; v_sku text; v_qty int; v_variants jsonb; i int; v jsonb; arr jsonb;
begin
  if (OLD.status in ('cancelled','refunded')) or (NEW.status not in ('cancelled','refunded')) then
    return new; -- 已经回补过或不是取消方向
  end if;
  for it in select * from jsonb_array_elements(NEW.items) loop
    v_id  = (it->>'id')::uuid; v_sku = coalesce(it->>'sku',''); v_qty = coalesce((it->>'qty')::int, 0);
    if v_id is not null and v_qty > 0 then
      if v_sku = '' then
        update public.products set stock = coalesce(stock,0) + v_qty where id = v_id;
      else
        select variants into v_variants from public.products where id = v_id;
        if v_variants is not null then
          arr = '[]'::jsonb;
          for i in 0 .. jsonb_array_length(v_variants) - 1 loop
            v = v_variants->i;
            if v->>'sku' = v_sku then
              v = jsonb_set(v, '{stock}', (coalesce((v->>'stock')::int,0) + v_qty)::text::jsonb);
            end if;
            arr = arr || v;
          end loop;
          update public.products set variants = arr where id = v_id;
        end if;
      end if;
    end if;
  end loop;
  return new;
end $$;
drop trigger if exists trg_shop_restore on public.shop_orders;
create trigger trg_shop_restore after update of status on public.shop_orders
  for each row execute function public.shop_restore_stock();

-- 拼团成团判定：任一成员订单变成已付款后检查（人数满 + 全员有效付款 → success）
create or replace function public.shop_check_group() returns trigger
language plpgsql security definer set search_path = public as $$
declare g_id uuid;
begin
  if NEW.group_id is null or NEW.status not in ('paid','shipped','completed') then
    return new;
  end if;
  g_id = NEW.group_id;
  update public.shop_groups g set status = 'success'
  where g.id = g_id and g.status = 'open'
    and g.joined >= g.target
    and not exists (
      select 1 from public.shop_group_members m
      join public.shop_orders o on o.id = m.order_id
      where m.group_id = g_id and o.status not in ('paid','shipped','completed')
    );
  return new;
end $$;
drop trigger if exists trg_shop_group on public.shop_orders;
create trigger trg_shop_group after update of status on public.shop_orders
  for each row execute function public.shop_check_group();

-- ═══════════ 第 9 步：下单 RPC（核心：服务端算价/验库存/防超卖/防重复）═══════════
create or replace function public.shop_place_order(
  p_items jsonb,        -- [{id, sku, qty}]  只传 id+sku+qty，价格服务端算
  p_address jsonb,      -- {name,phone,street,city,state,zip,country}
  p_note text default '',
  p_coupon text default '',
  p_group_id uuid default null,
  p_client_token text default null
) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  v_uid uuid := auth.uid();
  v_email text := coalesce(current_setting('request.jwt.claim.email', true), '');
  it jsonb; r record; v_sku text; v_qty int; v_price numeric; v_name text; v_image text; v_weight numeric := 0;
  snap jsonb := '[]'::jsonb; v_subtotal numeric := 0;
  v_ship jsonb; v_est numeric := 0; v_fee numeric := 2;
  v_disc numeric := 0; v_coupon record; v_total numeric;
  v_oid uuid; v_ono text; v_g record;
begin
  if v_uid is null then raise exception '请先登录'; end if;
  if p_items is null or jsonb_array_length(p_items) = 0 then raise exception '购物车是空的'; end if;
  if coalesce(p_address->>'name','') = '' or coalesce(p_address->>'street','') = '' then
    raise exception '收货地址不完整'; end if;

  -- 运费设置（后台 site_settings.shop_ship：{"service_fee":2,"flat_est":5}）
  select value into v_ship from public.site_settings where key = 'shop_ship';
  if v_ship is not null then
    v_fee = coalesce((v_ship->>'service_fee')::numeric, 2);
    v_est = coalesce((v_ship->>'flat_est')::numeric, 0);
  end if;

  -- 逐项验证：商品存在/在售/库存够，价格以数据库为准（淘宝式规格）
  for it in select * from jsonb_array_elements(p_items) loop
    v_qty = coalesce((it->>'qty')::int, 0);
    v_sku = coalesce(it->>'sku','');
    if v_qty < 1 then raise exception '数量不对'; end if;
    select * into r from public.products where id = (it->>'id')::uuid;
    if r.id is null or r.sold then raise exception '商品已下架'; end if;
    if v_sku = '' and r.variants is not null and jsonb_array_length(r.variants) > 0 then
      raise exception '请选择规格'; end if;
    if v_sku = '' then
      if coalesce(r.stock,0) < v_qty then raise exception '「%」库存不够了', r.name; end if;
      v_price = r.price::numeric; v_name = r.name; v_image = coalesce(r.image_url,'');
      update public.products set stock = coalesce(stock,0) - v_qty where id = r.id;
      v_weight = v_weight + coalesce(r.weight_oz,0) * v_qty;
    else
      declare v jsonb; found boolean := false; arr jsonb := '[]'::jsonb; i int;
      begin
        if r.variants is null then raise exception '商品没有这个规格'; end if;
        for i in 0 .. jsonb_array_length(r.variants) - 1 loop
          v = r.variants->i;
          if v->>'sku' = v_sku then
            found = true;
            if coalesce((v->>'stock')::int,0) < v_qty then
              raise exception '「% · %」库存不够了', r.name, coalesce(v->>'name', v_sku);
            end if;
            v_price = (v->>'price')::numeric; v_name = r.name || ' · ' || coalesce(v->>'name', v_sku);
            v_image = coalesce(v->>'images'->>0, r.image_url);
            v = jsonb_set(v, '{stock}', (coalesce((v->>'stock')::int,0) - v_qty)::text::jsonb);
            v_weight = v_weight + coalesce(r.weight_oz,0) * v_qty;
          end if;
          arr = arr || v;
        end loop;
        if not found then raise exception '规格不存在或已下架'; end if;
        update public.products set variants = arr where id = r.id;
      end;
    end if;
    snap = snap || jsonb_build_object('id', r.id, 'name', v_name, 'image', v_image,
      'sku', v_sku, 'price', v_price, 'qty', v_qty, 'weight_oz', coalesce(r.weight_oz,0));
    v_subtotal = v_subtotal + v_price * v_qty;
  end loop;

  -- 优惠券（数据库校验，客户端只传码）
  if coalesce(p_coupon,'') <> '' then
    select * into v_coupon from public.shop_coupons where code = p_coupon and active;
    if v_coupon is null then raise exception '优惠码无效'; end if;
    if v_coupon.starts_at is not null and now() < v_coupon.starts_at then raise exception '优惠活动还没开始'; end if;
    if v_coupon.ends_at is not null and now() > v_coupon.ends_at then raise exception '优惠活动已结束'; end if;
    if v_coupon.max_uses > 0 and v_coupon.used >= v_coupon.max_uses then raise exception '优惠码已被领完'; end if;
    if v_coupon.per_user > 0 and (select count(*) from public.shop_coupon_uses u where u.coupon_id = v_coupon.id and u.user_id = v_uid) >= v_coupon.per_user then
      raise exception '这个优惠码你已经用过了'; end if;
    if v_subtotal < v_coupon.min_spend then raise exception '没到优惠码的满减门槛'; end if;
    if v_coupon.tag <> '' and not exists (
      select 1 from jsonb_array_elements(snap) s
      join public.products pp on pp.id = (s->>'id')::uuid
      where pp.tag = v_coupon.tag) then
      raise exception '优惠码只适用于「%」商品', v_coupon.tag; end if;
    if v_coupon.type = 'percent' then v_disc = round(v_subtotal * v_coupon.value / 100, 2);
    else v_disc = least(v_coupon.value, v_subtotal); end if;
  end if;

  -- 拼团：占名额（原子）
  if p_group_id is not null then
    select * into v_g from public.shop_groups where id = p_group_id for update;
    if v_g is null then raise exception '拼团不存在'; end if;
    if v_g.status <> 'open' then raise exception '拼团已结束'; end if;
    if v_g.deadline < now() then raise exception '拼团已过期'; end if;
    if v_g.joined >= v_g.target then raise exception '拼团人数已满'; end if;
    if exists (select 1 from public.shop_group_members m where m.group_id = p_group_id and m.user_id = v_uid) then
      raise exception '你已经在这个拼团里啦'; end if;
    -- 拼团价覆盖（只支持单商品单规格的团）
    if jsonb_array_length(snap) = 1 then
      v_subtotal = v_g.price;
    end if;
  end if;

  v_total = v_subtotal - v_disc + v_est + v_fee;
  v_ono = 'YK' || to_char(now(),'YYMMDD') || '-' || upper(substr(replace(gen_random_uuid()::text,'-',''),1,6));

  insert into public.shop_orders (order_no, user_id, email, items, subtotal, discount, coupon_code,
      shipping_est, service_fee, total, status, address, note, group_id, client_token)
  values (v_ono, v_uid, v_email, snap, v_subtotal, v_disc, coalesce(p_coupon,''),
      v_est, v_fee, v_total, 'pending', p_address, coalesce(p_note,''), p_group_id, p_client_token)
  returning id into v_oid;

  if p_group_id is not null then
    insert into public.shop_group_members (group_id, user_id, nickname, order_id)
    values (p_group_id, v_uid, coalesce((select nickname from public.shop_profiles where id = v_uid), split_part(v_email,'@',1)), v_oid);
    update public.shop_groups set joined = joined + 1 where id = p_group_id;
  end if;

  if v_coupon is not null then
    insert into public.shop_coupon_uses (coupon_id, user_id, order_id) values (v_coupon.id, v_uid, v_oid);
    update public.shop_coupons set used = used + 1 where id = v_coupon.id;
  end if;

  insert into public.shop_order_logs (order_id, actor, action, detail)
  values (v_oid, 'buyer', 'create', '下单 ' || v_ono);

  return jsonb_build_object('order_id', v_oid, 'order_no', v_ono, 'total', v_total,
    'subtotal', v_subtotal, 'discount', v_disc, 'shipping_est', v_est, 'service_fee', v_fee);
end $$;

-- 取消订单（买家）：释放库存（触发器自动回补）、退出拼团释放名额
create or replace function public.shop_cancel_order(p_order_id uuid) returns text
language plpgsql security definer set search_path = public as $$
declare o record;
begin
  select * into o from public.shop_orders where id = p_order_id and user_id = auth.uid() for update;
  if o is null then raise exception '订单不存在'; end if;
  if o.status not in ('pending','reviewing','rejected') then
    raise exception '当前状态不能取消，已付款订单请走售后'; end if;
  update public.shop_orders set status = 'cancelled' where id = p_order_id;
  if o.group_id is not null then
    delete from public.shop_group_members where group_id = o.group_id and user_id = o.user_id;
    update public.shop_groups set joined = greatest(joined - 1, 0) where id = o.group_id;
  end if;
  return 'cancelled';
end $$;

-- 过期拼团兜底（任何人可触发，只会把真过期的团标记失败）
create or replace function public.shop_finalize_groups() returns void
language sql security definer set search_path = public as $$
  update public.shop_groups set status = 'failed'
  where status = 'open' and deadline < now();
$$;
grant execute on function public.shop_finalize_groups() to anon, authenticated;
grant execute on function public.shop_place_order(jsonb, jsonb, text, text, uuid, text) to authenticated;
grant execute on function public.shop_cancel_order(uuid) to authenticated;

  using (bucket_id = 'pay' and ((storage.foldername(name))[1] = auth.uid()::text or public.shop_is_admin()));
